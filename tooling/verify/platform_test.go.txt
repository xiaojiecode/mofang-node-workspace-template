package pagecompiler_test

import (
    "context"
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "testing"
    "mofang/pkg/jsprogram"
    "mofang/pkg/nodeabi"
    "mofang/services/workflow/internal/pagecompiler"
)

// TestEngineeringTemplate uses the actual Goja module/stage and page compilers; no database or external Tool.
func TestEngineeringTemplate(t *testing.T) {
    root := os.Getenv("MF_TEMPLATE_REPO")
    if root == "" { t.Fatal("missing MF_TEMPLATE_REPO") }
    entries,err:=os.ReadDir(filepath.Join(root,"dist"));if err!=nil{t.Fatal(err)}
    for _,entry:=range entries { if !entry.IsDir(){continue}; t.Run(entry.Name(),func(t *testing.T){
        out:=filepath.Join(root,"dist",entry.Name())
        read:=func(name string)string{t.Helper();value,err:=os.ReadFile(filepath.Join(out,name));if err!=nil{t.Fatal(err)};return string(value)}
        ctx:=context.Background()
        for _,surface:=range []string{"settings","business"}{
            source:=pagecompiler.Sources{Surface:surface,ViewSource:read(surface+".js"),SDKVersion:"1"}
            if surface=="business"{source.ProgramSource=read("pageProgram.js");source.SupportedContextSchemas=[]string{"mf.node-context/1"}}
            if _,err:=pagecompiler.New().Compile(ctx,source);err!=nil{t.Fatalf("%s page compile: %v",surface,err)}
        }
        module,err:=jsprogram.CompileModule(ctx,read("program.js"));if err!=nil{t.Fatal(err)}
        facts:=nodeabi.NodeFacts{RunID:"fixture-run",NodeInstanceID:"fixture-node",SessionRevision:"1",Status:nodeabi.Status{Run:"running",Session:"running",Closure:"open"},Gate:nodeabi.Gate{CanHandle:true},Policy:nodeabi.Policy{Mode:"manual"}}
        config:=json.RawMessage(`{}`)
        state,err:=module.Initialize(ctx,nodeabi.InitializeInput{SourceHash:module.SourceHash(),Config:config,Facts:facts});if err!=nil{t.Fatal(err)}
        allocated,sequence:=0,0
        send:=func(kind string,payload any)nodeabi.HandleResult{
            t.Helper();sequence++
            raw,err:=json.Marshal(payload);if err!=nil{t.Fatal(err)}
            configHash,err:=nodeabi.Digest(config);if err!=nil{t.Fatal(err)};factsHash,err:=nodeabi.Digest(facts);if err!=nil{t.Fatal(err)}
            input:=nodeabi.HandleInput{Basis:nodeabi.Basis{SourceHash:module.SourceHash(),SessionRevision:"1",ConfigDigest:configHash,FactsDigest:factsHash},State:state,Config:config,Facts:facts,Event:nodeabi.EventEnvelope{ID:fmt.Sprintf("fixture-event-%d",sequence),Sequence:fmt.Sprint(sequence),Type:kind,CreatedAt:"2026-10-23T00:00:00Z",Payload:raw}}
            result,err:=module.Handle(ctx,input,jsprogram.ModuleHost{AllocateID:func(kind string)(string,error){allocated++;return fmt.Sprintf("fixture-%s-%d",kind,allocated),nil}});if err!=nil{t.Fatal(err)}
            if result.Kind!="apply"{t.Fatalf("handle %s: %+v",kind,result)};state=*result.NextState;return result
        }
        send("start",map[string]any{})
        fixture,err:=os.ReadFile(filepath.Join(root,"nodes",entry.Name(),"fixtures/input.json"));if err!=nil{t.Fatal(err)}
        result:=send("input",map[string]any{"deliveryId":"fixture-delivery","values":json.RawMessage(fixture)})
        if entry.Name()!="selection"&&len(result.Plan.Commands)!=1{t.Fatalf("input command count: %d",len(result.Plan.Commands))}
        if len(result.Plan.Commands)==1&&result.Plan.Commands[0].Kind=="task.spawn"{
            command:=result.Plan.Commands[0]
            task,err:=module.Task(command.Handler);if err!=nil{t.Fatal(err)}
            host:=jsprogram.Context{Input:command.Input,State:json.RawMessage(`{}`),Call:func(context.Context,string,json.RawMessage)(json.RawMessage,error){return nil,fmt.Errorf("external calls are not authorized")}}
            transition,err:=task.RunStep(ctx,task.Entry(),host);if err!=nil{t.Fatal(err)}
            if transition.AwaitUser!=nil{
                host.State=transition.AwaitUser.Data;host.Answer=json.RawMessage(`{"approved":true}`)
                transition,err=task.RunStep(ctx,transition.AwaitUser.NextStep,host);if err!=nil{t.Fatal(err)}
            }
            if len(transition.Output)==0{t.Fatal("fixture stage did not produce output")}
            facts.InputClosed=true
            result=send("task-result",map[string]any{"taskId":command.ID,"handler":command.Handler,"inputRefs":[]string{"fixture-delivery"},"status":"succeeded","output":json.RawMessage(transition.Output)})
        }else if entry.Name()=="selection"{
            facts.InputClosed=true
            send("input-closed",map[string]string{"reason":"completed"})
            result=send("action",map[string]any{"name":"submit","payload":map[string]any{"keys":[]string{"fixture-delivery"}},"candidates":[]any{},"actor":map[string]string{"userId":"fixture-user","workspaceId":"fixture-workspace","role":"user"}})
        }
        if len(result.Plan.Commands)!=1||result.Plan.Commands[0].Kind!="candidate.append"||len(result.Plan.Commands[0].InputRefs)!=1||result.Plan.Commands[0].InputRefs[0]!="fixture-delivery"{t.Fatalf("candidate/source lost: %+v",result)}
        configHash,_:=nodeabi.Digest(config);factsHash,_:=nodeabi.Digest(facts)
        projection,err:=module.Project(ctx,nodeabi.ProjectInput{Basis:nodeabi.Basis{SourceHash:module.SourceHash(),SessionRevision:"1",ConfigDigest:configHash,FactsDigest:factsHash},State:state,Config:config,Facts:facts});if err!=nil||projection.Kind!="ready"{t.Fatalf("projection: %+v %v",projection,err)}
    }) }
}
