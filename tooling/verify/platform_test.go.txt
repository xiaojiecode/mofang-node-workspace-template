package pagecompiler_test

import (
	"context"
	"encoding/json"
	"os"
	"path/filepath"
	"testing"

	"mofang/pkg/jsprogram"
	"mofang/pkg/nodeabi"
	"mofang/services/workflow/internal/pagecompiler"
)

// TestEngineeringTemplate 使用真实 Goja 模块与页面编译器，不访问数据库或外部能力。
func TestEngineeringTemplate(t *testing.T) {
	root := os.Getenv("MF_TEMPLATE_REPO")
	if root == "" {
		t.Fatal("missing MF_TEMPLATE_REPO")
	}
	var workspace struct {
		Kind string `json:"kind"`
	}
	workspaceSource, err := os.ReadFile(filepath.Join(root, ".agents/upstream.json"))
	if err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(workspaceSource, &workspace); err != nil {
		t.Fatal(err)
	}
	if workspace.Kind != "template" && workspace.Kind != "project" {
		t.Fatal("unsupported workspace kind")
	}
	entries, err := os.ReadDir(filepath.Join(root, "dist"))
	if err != nil {
		t.Fatal(err)
	}
	tested := 0
	for _, entry := range entries {
		if !entry.IsDir() {
			continue
		}
		tested++
		t.Run(entry.Name(), func(t *testing.T) {
			out := filepath.Join(root, "dist", entry.Name())
			read := func(name string) string {
				t.Helper()
				value, err := os.ReadFile(filepath.Join(out, name))
				if err != nil {
					t.Fatal(err)
				}
				return string(value)
			}
			ctx := context.Background()
			for _, surface := range []string{"settings", "business"} {
				source := pagecompiler.Sources{Surface: surface, ViewSource: read(surface + ".js"), SDKVersion: "1"}
				if surface == "business" {
					source.ProgramSource = read("pageProgram.js")
					source.SupportedContextSchemas = []string{"mf.node-context/1"}
				}
				if _, err := pagecompiler.New().Compile(ctx, source); err != nil {
					t.Fatalf("%s page compile: %v", surface, err)
				}
			}
			var manifest struct {
				Interface string `json:"interface"`
			}
			if err := json.Unmarshal([]byte(read("manifest.json")), &manifest); err != nil {
				t.Fatal(err)
			}
			if manifest.Interface == "panels" {
				return
			}
			module, err := jsprogram.CompileModule(ctx, read("program.js"))
			if err != nil {
				t.Fatal(err)
			}
			facts := nodeabi.NodeFacts{RunID: "fixture-run", NodeInstanceID: "fixture-node", SessionRevision: "1", Status: nodeabi.Status{Run: "running", Session: "running", Closure: "open"}, Gate: nodeabi.Gate{CanHandle: true}, Policy: nodeabi.Policy{Mode: "manual"}}
			config := json.RawMessage(`{}`)
			state, err := module.Initialize(ctx, nodeabi.InitializeInput{SourceHash: module.SourceHash(), Config: config, Facts: facts})
			if err != nil {
				t.Fatal(err)
			}
			// 不按目录名猜测业务行为；执行行为由节点自己的完整 Goja fixture 负责。
			configHash, err := nodeabi.Digest(config)
			if err != nil {
				t.Fatal(err)
			}
			factsHash, err := nodeabi.Digest(facts)
			if err != nil {
				t.Fatal(err)
			}
			projection, err := module.Project(ctx, nodeabi.ProjectInput{Basis: nodeabi.Basis{SourceHash: module.SourceHash(), SessionRevision: "1", ConfigDigest: configHash, FactsDigest: factsHash}, State: state, Config: config, Facts: facts})
			if err != nil {
				t.Fatal(err)
			}
			if projection.Kind != "ready" && projection.Kind != "unavailable" {
				t.Fatalf("invalid projection contract: %+v", projection)
			}
			t.Logf("node ABI preflight: projection=%s; business fixture is verified separately", projection.Kind)
		})
	}
	if tested == 0 {
		t.Fatal("没有可验证节点，请在独立工程中通过 node:new 创建骨架并 build 后重试")
	}
}
